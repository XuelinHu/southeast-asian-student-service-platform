# 面向国际留学生的信息管理系统V1.0源代码

## 1 源代码属性结构说明

### 1.1 项目技术栈

本软件采用 H5、Node.js、Express 和 MySQL 技术栈实现。前端由 HTML、CSS 和 JavaScript 组成，提供学生端双语 H5 页面以及教师端、管理端的横向管理界面。后端由 Node.js 和 Express 提供 REST API，包含登录认证、学生档案、教师管理、成绩适配、统计分析和风险查询等接口。数据库使用 MySQL，保存用户、学生、教师、班级、国籍、住宿、签证、学习记录、外部学校、成绩规则和成绩记录等业务数据。

### 1.2 源码根目录

源码根目录为当前项目目录。源码稿只选入与本系统运行直接相关的自研业务源码、完整后台代码、前端页面源码、数据库脚本、业务种子脚本和必要配置文件，不选入第三方依赖、锁文件、截图、PDF、缓存文件、说明书、Markdown 文档正文或软著材料生成脚本。

### 1.3 一级目录划分

- `src`：Node.js 后端源码，包含服务入口、路由、中间件、数据库连接和成绩适配服务。
- `public`：H5 前台和管理端静态资源，包含页面结构、样式和交互逻辑。
- `database`：MySQL 建库建表和基础数据脚本。
- `scripts`：软著材料生成脚本，不属于系统运行源码，本源码稿不纳入。
- `docs`：软著材料输出目录，不纳入源码正文。

### 2.5 文件：`src/middleware/auth.js`

文件职责：JWT 鉴权中间件，按角色限制接口访问。

所属模块：认证与权限。

```javascript
0001  const jwt = require('jsonwebtoken');
0002  
0003  function auth(requiredRoles = []) {
0004    return (req, res, next) => {
0005      const header = req.headers.authorization || '';
0006      const token = header.startsWith('Bearer ') ? header.slice(7) : '';
0007  
0008      if (!token) {
0009        return res.status(401).json({ message: 'Missing authorization token' });
0010      }
0011  
0012      try {
0013        const payload = jwt.verify(token, process.env.JWT_SECRET || 'dev_secret');
0014        if (requiredRoles.length && !requiredRoles.includes(payload.role)) {
0015          return res.status(403).json({ message: 'Permission denied' });
0016        }
0017        req.user = payload;
0018        next();
0019      } catch (error) {
0020        return res.status(401).json({ message: 'Invalid or expired token' });
0021      }
0022    };
0023  }
0024  
0025  module.exports = auth;
```

### 2.6 文件：`src/routes/auth.js`

文件职责：登录接口路由，完成账号校验、密码校验和令牌签发。

所属模块：认证与权限。

```javascript
0001  const express = require('express');
0002  const bcrypt = require('bcryptjs');
0003  const jwt = require('jsonwebtoken');
0004  const pool = require('../config/db');
0005  
0006  const router = express.Router();
0007  
0008  router.post('/login', async (req, res, next) => {
0009    try {
0010      const { username, password } = req.body;
0011      const [rows] = await pool.execute(
0012        'SELECT id, username, password_hash, role, display_name, status FROM users WHERE username = :username LIMIT 1',
0013        { username }
0014      );
0015  
0016      const user = rows[0];
0017      if (!user || user.status !== 'active') {
0018        return res.status(401).json({ message: 'Invalid username or password' });
0019      }
0020  
0021      const ok = await bcrypt.compare(password, user.password_hash);
0022      if (!ok) {
0023        return res.status(401).json({ message: 'Invalid username or password' });
0024      }
0025  
0026      const token = jwt.sign(
0027        { id: user.id, username: user.username, role: user.role, name: user.display_name },
0028        process.env.JWT_SECRET || 'dev_secret',
0029        { expiresIn: process.env.JWT_EXPIRES_IN || '8h' }
0030      );
0031  
0032      res.json({
0033        token,
0034        user: {
0035          id: user.id,
0036          username: user.username,
0037          role: user.role,
0038          displayName: user.display_name
0039        }
0040      });
0041    } catch (error) {
0042      next(error);
0043    }
0044  });

0046  module.exports = router;
```

所属模块：成绩适配模块。

```javascript
0001  const LETTER_TO_PERCENT = {
0002    A: 95,
0003    'A-': 90,
0004    B: 85,
0005    'B-': 80,
0006    C: 75,
0007    'C-': 70,
0008    D: 65,

0012  function gpaToPercent(value, scale) {
0013    const numeric = Number(value);
0014    const max = Number(scale || 4);
0015    if (Number.isNaN(numeric) || max <= 0) {
0016      throw new Error('Invalid GPA value');
0017    }
0018    return Math.round((numeric / max) * 100);
0019  }
0020  
0021  function toGpa4(percent) {
0022    if (percent >= 90) return 4.0;
0023    if (percent >= 85) return 3.7;
0024    if (percent >= 80) return 3.3;
0025    if (percent >= 75) return 3.0;
0026    if (percent >= 70) return 2.7;
0027    if (percent >= 65) return 2.3;
0028    if (percent >= 60) return 2.0;
0029    return 0;
0030  }
0031  
0032  function adaptGrade(input, rule = {}) {
0033    const type = String(input.originalType || '').toLowerCase();
0034    let percent;
0035  
0036    if (type === 'percent') {
0037      percent = Number(input.originalGrade);
0038    } else if (type === 'letter') {
0039      percent = LETTER_TO_PERCENT[String(input.originalGrade).trim().toUpperCase()];
0040    } else if (type === 'gpa') {
0041      percent = gpaToPercent(input.originalGrade, rule.gpaScale || input.gpaScale || 4);
0042    } else if (type === 'pass_fail') {
0043      percent = String(input.originalGrade).toLowerCase() === 'pass' ? 75 : 50;
0044    } else {
0045      throw new Error('Unsupported grade type');
0046    }
0047  
0048    if (typeof percent !== 'number' || Number.isNaN(percent) || percent < 0 || percent > 100) {
0049      throw new Error('Adapted grade is out of range');
0050    }
0051  
0052    const creditRatio = Number(rule.creditRatio || input.creditRatio || 1);
0053    return {
0054      adaptedScore: Math.round(percent),
0055      adaptedGpa: toGpa4(percent),
0056      adaptedCredit: Number((Number(input.originalCredit || 0) * creditRatio).toFixed(2)),
0057      passed: percent >= Number(rule.passScore || 60),
0058      ruleVersion: rule.version || 'default-v1'
0059    };
0060  }
```
