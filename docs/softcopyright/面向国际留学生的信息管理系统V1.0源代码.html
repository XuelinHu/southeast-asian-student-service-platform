# 面向国际留学生的信息管理系统 V1.0

前台使用 H5，后台使用 Node.js + Express，数据库使用 MySQL。

## 功能范围

- 学生登录与学生端中英双语界面
- 学生档案、国籍、班级、住宿、签证、学习情况基础数据模型
- 教师管理、教师班级管理、教师成绩管理入口
- 多国家、多学校成绩对接与成绩适配接口
- 成绩原始值、适配值、GPA、学分、规则版本、审核状态和异常原因留痕
- 管理端统计分析接口

## 数据库配置

默认连接配置：

host: *** port: 3306 user: root password: *** database: international_student_platform


## 启动步骤

1. 安装依赖：

npm install


2. 创建本地环境配置：

cp .env.example .env


3. 初始化数据库：

mysql -h *** -P 3306 -u root -p < database/schema.sql


也可以使用内置 Node 脚本初始化，不依赖本机 `mysql` 命令：

npm run db:init


4. 写入业务关联测试数据。该脚本会让每张业务表不少于 100 条记录，并保持用户、学生、教师、班级、住宿、签证、学习、外部学校、成绩规则和成绩记录之间的业务关联：

npm run db:seed


5. 启动服务：

npm start


访问地址：

http://***:3000


## 演示账号

三个演示账号的初始密码均为 `***`。

admin teacher01 student01 teacher001 student001


## 核心接口

- `POST /api/auth/login` 登录
- `GET /api/students/me` 学生查看本人档案
- `GET /api/students` 教师/管理员查看学生列表
- `GET /api/teachers` 管理员查看教师列表
- `POST /api/grades/adapt` 教师/管理员提交外部成绩并执行适配
- `GET /api/grades/student/:studentId` 查看学生成绩适配记录
- `PATCH /api/grades/:id/audit` 管理员审核成绩
- `GET /api/management/dashboard` 统计分析
- `GET /api/management/risks` 风险查询，聚合签证临期、成绩适配异常、低分风险、学习预警和住宿异常

function applyLanguage() {
  document.documentElement.lang = lang === 'zh' ? 'zh-CN' : 'en';
  document.querySelectorAll('[data-i18n]').forEach((node) => {

async function api(path, options = {}) {
  const headers = {
    'Content-Type': 'application/json',
    ...(options.headers || {})
  };
  if (session?.token) headers.Authorization = `Bearer ${session.token}`;
  const response = await fetch(path, { ...options, headers });
  if (!response.ok) {
    const body = await response.json().catch(() => ({}));
    throw new Error(body.message || response.statusText);
  }
  return response.json();
}

function roleTabs(role) {
  if (role === 'student') return ['profile', 'grades', 'learning'];
  if (role === 'teacher') return ['dashboard', 'students', 'grades', 'risks'];
  return ['dashboard', 'students', 'teacher', 'grades', 'risks'];
}

function renderTabs() {
  tabs.innerHTML = roleTabs(session.user.role)
    .map((tab) => `<button class="tab-button ${tab === currentTab ? 'active' : ''}" data-tab="${tab}">${t(tab)}</button>`)
    .join('');
  tabs.querySelectorAll('button').forEach((button) => {
    button.addEventListener('click', () => {
      currentTab = button.dataset.tab;
      renderApp();
    });
  });
}

function metric(label, value, tone = '') {
  return `<div class="metric"><span class="muted">${label}</span><strong class="${tone}">${value}</strong></div>`;
}

function table(headers, rows) {
  if (!rows.length) return `<p class="muted">${t('noData')}</p>`;
  return `
    <table class="data-table">
      <thead><tr>${headers.map((item) => `<th>${item}</th>`).join('')}</tr></thead>
      <tbody>${rows.join('')}</tbody>
    </table>
  `;
}

    metric(t('status'), grades[0]?.adapt_status || '-')

const jwt = require('jsonwebtoken');

function auth(requiredRoles = []) {
  return (req, res, next) => {
    const header = req.headers.authorization || '';
    const token = header.startsWith('Bearer ') ? header.slice(7) : '';

    if (!token) {
      return res.status(401).json({ message: 'Missing authorization token' });
    }

    try {
      const payload = jwt.verify(token, process.env.JWT_SECRET || '***');
      if (requiredRoles.length && !requiredRoles.includes(payload.role)) {
        return res.status(403).json({ message: 'Permission denied' });
      }
      req.user = payload;
      next();
    } catch (error) {
      return res.status(401).json({ message: 'Invalid or expired token' });
    }
  };
}

module.exports = auth;

const router = express.Router();

router.post('/login', async (req, res, next) => {
  try {
    const { username, password } = req.body;
    const [rows] = await pool.execute(
      'SELECT id, username, password_hash, role, display_name, status FROM users WHERE username = :username LIMIT 1',
      { username }
    );

    const user = rows[0];
    if (!user || user.status !== 'active') {
      return res.status(401).json({ message: 'Invalid username or password' });
    }

    const ok = await bcrypt.compare(password, user.password_hash);
    if (!ok) {
      return res.status(401).json({ message: 'Invalid username or password' });
    }

    const token = jwt.sign(
      { id: user.id, username: user.username, role: user.role, name: user.display_name },
      process.env.JWT_SECRET || '***',
      { expiresIn: process.env.JWT_EXPIRES_IN || '8h' }
    );

    res.json({
      token,
      user: {
        id: user.id,
        username: user.username,
        role: user.role,
        displayName: user.display_name
      }
    });
  } catch (error) {

const LETTER_TO_PERCENT = {
  A: 95,
  'A-': 90,
  B: 85,
  'B-': 80,
  C: 75,
  'C-': 70,
  D: 65,
  F: 50
};

function gpaToPercent(value, scale) {
  const numeric = Number(value);
  const max = Number(scale || 4);
  if (Number.isNaN(numeric) || max <= 0) {
    throw new Error('Invalid GPA value');
  }
  return Math.round((numeric / max) * 100);
}

function toGpa4(percent) {
  if (percent >= 90) return 4.0;
  if (percent >= 85) return 3.7;
  if (percent >= 80) return 3.3;
  if (percent >= 75) return 3.0;
  if (percent >= 70) return 2.7;
  if (percent >= 65) return 2.3;
  if (percent >= 60) return 2.0;
  return 0;
}

function adaptGrade(input, rule = {}) {
  const type = String(input.originalType || '').toLowerCase();
  let percent;

  if (type === 'percent') {
    percent = Number(input.originalGrade);
  } else if (type === 'letter') {
    percent = LETTER_TO_PERCENT[String(input.originalGrade).trim().toUpperCase()];
  } else if (type === 'gpa') {
    percent = gpaToPercent(input.originalGrade, rule.gpaScale || input.gpaScale || 4);
  } else if (type === 'pass_fail') {
    percent = String(input.originalGrade).toLowerCase() === 'pass' ? 75 : 50;
  } else {
    throw new Error('Unsupported grade type');
  }

  if (typeof percent !== 'number' || Number.isNaN(percent) || percent < 0 || percent > 100) {
    throw new Error('Adapted grade is out of range');
  }

  const creditRatio = Number(rule.creditRatio || input.creditRatio || 1);
  return {
    adaptedScore: Math.round(percent),
    adaptedGpa: toGpa4(percent),
    adaptedCredit: Number((Number(input.originalCredit || 0) * creditRatio).toFixed(2)),
    passed: percent >= Number(rule.passScore || 60),
    ruleVersion: rule.version || 'default-v1'
  };
}

const jwt = require('jsonwebtoken');

function auth(requiredRoles = []) {
  return (req, res, next) => {
    const header = req.headers.authorization || '';
    const token = header.startsWith('Bearer ') ? header.slice(7) : '';

    if (!token) {
      return res.status(401).json({ message: 'Missing authorization token' });
    }

    try {
      const payload = jwt.verify(token, process.env.JWT_SECRET || '***');
      if (requiredRoles.length && !requiredRoles.includes(payload.role)) {
        return res.status(403).json({ message: 'Permission denied' });
      }
      req.user = payload;
      next();
    } catch (error) {
      return res.status(401).json({ message: 'Invalid or expired token' });
    }
  };
}

module.exports = auth;

const LETTER_TO_PERCENT = {
  A: 95,
  'A-': 90,
  B: 85,
  'B-': 80,
  C: 75,
  'C-': 70,
  D: 65,
  F: 50
};

function gpaToPercent(value, scale) {
  const numeric = Number(value);
  const max = Number(scale || 4);
  if (Number.isNaN(numeric) || max <= 0) {
    throw new Error('Invalid GPA value');
  }
  return Math.round((numeric / max) * 100);
}

function toGpa4(percent) {
  if (percent >= 90) return 4.0;
  if (percent >= 85) return 3.7;
  if (percent >= 80) return 3.3;
  if (percent >= 75) return 3.0;
  if (percent >= 70) return 2.7;
  if (percent >= 65) return 2.3;
  if (percent >= 60) return 2.0;
  return 0;
}

function adaptGrade(input, rule = {}) {
  const type = String(input.originalType || '').toLowerCase();
  let percent;

  if (type === 'percent') {
    percent = Number(input.originalGrade);
  } else if (type === 'letter') {
    percent = LETTER_TO_PERCENT[String(input.originalGrade).trim().toUpperCase()];
  } else if (type === 'gpa') {
    percent = gpaToPercent(input.originalGrade, rule.gpaScale || input.gpaScale || 4);
  } else if (type === 'pass_fail') {
    percent = String(input.originalGrade).toLowerCase() === 'pass' ? 75 : 50;
  } else {
    throw new Error('Unsupported grade type');
  }

  if (typeof percent !== 'number' || Number.isNaN(percent) || percent < 0 || percent > 100) {
    throw new Error('Adapted grade is out of range');
  }

  const creditRatio = Number(rule.creditRatio || input.creditRatio || 1);
  return {
    adaptedScore: Math.round(percent),
    adaptedGpa: toGpa4(percent),
    adaptedCredit: Number((Number(input.originalCredit || 0) * creditRatio).toFixed(2)),
    passed: percent >= Number(rule.passScore || 60),
    ruleVersion: rule.version || 'default-v1'
  };
}

module.exports = { adaptGrade };
